\cs_new:Npn \guj_digit:n #1 
{ 
    \int_case:nnF {#1} 
    { 
        {0}{૦} 
        {1}{૧} 
        {2}{૨} 
        {3}{૩} 
        {4}{૪} 
        {5}{૫} 
        {6}{૬} 
        {7}{૭} 
        {8}{૮} 
        {9}{૯} 
    } 
    {#1} 
} 

\cs_new:Npn \guj_digits:n #1 
{ 
    \tl_map_function:nN { #1 } \guj_digit:n 
} 

\NewExpandableDocumentCommand{\gujaratiarabic}{m} 
{ 
    \exp_args:Ne \guj_digits:n { \int_to_arabic:n {#1} } 
}
